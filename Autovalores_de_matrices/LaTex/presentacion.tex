\section{Introducción}

\begin{frame}{Cerebro crítico e IA}
    Los nuevos métodos de electrofisiología y neuroimagen han mostrado que es probable que el cerebro trabaje en un régimen cercano a la {\color{blue}criticidad} \cite{beggs2003}, donde se ha mostrado que la {\color{blue}computación neuromórfica} obtiene una {\color{blue}capacidad de análisis} óptima \cite{bertschinger2004}.
    \vfill
    \begin{columns}[c]
        \hfill
        \begin{column}{0.45\textwidth}
            Las claves para entender este fenómeno son:
            \begin{itemize}
                \item Criticidad Autoorganizada (SOC)
                \item Criticidad dinámica 
                \item Modelos de aprendizaje
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{0.45\textwidth}
            \includegraphics[width=.6\textwidth]{Figures/sensores.png}

            Fuente de la imagen \cite{beggs2003}
        \end{column}
    \end{columns}

\end{frame}

\begin{frame}{Estabilidad y funcionalidad}
    \textbf{Pregunta:} ¿Cómo es posible que las {\color{blue}sinapsis} entre neuronas cambien y se adapten para evitar inestabilidades y picos extremos de actividad, pero que sean capaces de almacenar {\color{blue}recuerdos} y posean funcionalidad?

    \begin{center}
        \includegraphics[width=0.6\textwidth]{Figures/Explosion llave inglesa.png}
    \end{center}

    \textbf{Respuesta:} Desacoplar la matriz de conexiones en la parte {\color{blue}simétrica}, encargada de la {\color{blue}estabilidad} y la {\color{blue}antisimétrica} para la {\color{blue}funcionalidad}.
\end{frame}

\section{Modelo de Red Neuronal}
\begin{frame}{Modelo de Red Neuronal}
    La actividad de \(N\) neuronas, vector \(\mathbf{x}\), y las conexiones entre ellas, \(W\), se modelan mediante las siguientes ecuaciones diferenciales 
    \vfill
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{eqbox}
                \begin{equation}
                    \dot{\mathbf{x}} = W\mathbf{x}
                \end{equation}           
            \end{eqbox}

        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{eqbox}
                \begin{equation}
                    \dot{W} = \Delta_H + \Delta_L
                \end{equation}        
            \end{eqbox}
        \end{column}
    \end{columns}
    \vfill
    \begin{itemize}
        \item \(\Delta_H\): Término de homeostasis encargado de estabilizar la actividad de la red.
        \item \(\Delta_L\): Término de aprendizaje (\emph{learning}) encargado de la funcionalidad (almacenar memorias).
    \end{itemize}
\end{frame}

\begin{frame}{Estabilidad y Autovalores}
    La estabilidad de la red de neuronas depende de la parte real de los autovalores de la matriz de conexiones \(W\).



    \begin{equation*}
        S = \frac{W + W^T}{2} \rightarrow \text{Re}(\lambda) \qquad \qquad A = \frac{W - W^T}{2} \rightarrow \text{Im}(\lambda)
    \end{equation*}

 Cada una de las componentes de los autovalores está asociada a una de las componentes de la matriz de conexiones.
\end{frame}

\begin{frame}{Estabilidad y Autovalores}
    \begin{columns}
        \begin{column}{0.38\textwidth}
            La estabilidad del sistema depende de \(\text{Re}(\lambda)\)
            \vspace{0.2in}
            \begin{itemize}
                \item \textbf{\color{blue}Estable:} \(\text{Re}(\lambda) < 0\)
                \item \textbf{\color{black}Crítico:} \(\text{Re}(\lambda) \approx 0\)
                \item \textbf{\color{red}Inestable:} \(\text{Re}(\lambda) > 0\)
            \end{itemize}
        \end{column}
        \begin{column}{0.6\textwidth}
            \centering
            \includegraphics[width=\textwidth]{Figures/diagrama estabilidad anotado.pdf}
            \vspace*{-0.1in}
            Plano complejo
        \end{column}
    \end{columns}
\end{frame}
\section{Aprendizaje Antihebbiano}
\begin{frame}{Aprendizaje Antihebbiano}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \centering
            \textbf{\Large Hebbiano}
            \begin{eqbox}
                \begin{equation*}
                    \Delta_H \propto \mathbf{xx}^T
                \end{equation*}
            \end{eqbox}

            {\color{red}Refuerza} las conexiones entre neuronas que disparan juntas.\\
            \(\Downarrow\)\\
            Sistema {\color{red}divergente e inestable}.
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \textbf{\Large Antihebbiano}
            \begin{eqbox}
                \begin{equation*}
                    \Delta_H = \alpha(\mathbb{1} - \mathbf{xx}^T)
                \end{equation*}
            \end{eqbox}
            {\color{blue}Debilita} las conexiones entre neuronas que disparan juntas.\\
            \(\Downarrow\)\\
            Sistema {\color{blue}críticamente estable}.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Concentración de autovalores}
    \centering
    \includegraphics[width=\textwidth]{Figures/Evolucion coloreada.pdf}

    Como el término antihebbiano es simétrico, la parte imaginaria de los autovalores apenas es afectada, mientras que la parte real de los autovalores se coloca en torno a 0, en la región dinámicamente crítica.
\end{frame}

\begin{frame}{Desacople de Modos}
    \centering

    La {\color{blue}actividad} de la red puede desacoplarse en {\color{blue}modos independientes} en forma de osciladores unidimensionales. Para dimensión baja la aproximación es excelente.
\end{frame}

\begin{frame}{Desacople de Modos}

\end{frame}

\begin{frame}{Criticidad Autoorganizada}

\begin{frame}{Criticidad Autoorganizada}
    Los sistemas con {\color{blue}criticidad autoorganizada} son inherentemente de {\color{blue}no equilibrio} y {\color{blue}dinámicos}, como

    \begin{itemize}
        \item Terremotos
        \item Incendios forestales
        \item Avalanchas
    \end{itemize}

    Dichos sistemas generan eventos {\color{blue}sin escala característica}, en los que se dan sucesos tanto mínimos como extremos. Vienen dados por distribuciones de {\color{blue}leyes de potencias} \cite{bak1987}.

    \begin{align}
        \text{Tamaño de las avalanchas} \quad P(S) &\propto S^{-\tau}\\
        \text{Duración de las avalanchas} \quad P(T) &\propto T^{-\nu}
    \end{align}
\end{frame}

\begin{frame}{Ausencia de Criticidad Autoorganizada}
    \begin{columns}
        \begin{column}{0.35\textwidth}
            Se muestra la CCDF
            \begin{equation*}
                CCDF(S) = P(x>S)
            \end{equation*}
            Si hubiese SOC la CCDF debería seguir una ley de potencias. Se demuestra que:
            \begin{itemize}
                \item Existe una escala característica
                \item No hay \emph{scaling}
            \end{itemize}
        \end{column}
        \begin{column}{0.63\textwidth}
        \centering
        \includegraphics[width=\textwidth]{Figures/powerlaw.pdf}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Ausencia de Criticidad Autoorganizada}
        \begin{columns}
        \begin{column}{0.35\textwidth}
            \centering
            Si hubiese SOC deberían aparecer correlaciones a {\color{blue}todas las escalas}.
            El espectro de la matriz de covarianza no presenta {\color{blue}leyes de potencias}.\\ 
            \(\Downarrow\)\\
             Neuronas completamente {\color{blue}decorrelacionadas}. No hay SOC.
        \end{column}
        \begin{column}{0.63\textwidth}

\end{frame}

\begin{frame}{Aprendizaje Antisimétrco}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \centering
            \textbf{\Large Homeostasis}
            \begin{eqbox}
                \begin{equation*}
                    \Delta_H = \alpha(\mathbb{1} - \mathbf{xx}^T)
                \end{equation*}
            \end{eqbox}

            El término de homeostasis es {\color{blue}simétrico}, requisito mínimo para modificar la {\color{blue}estabilidad}.
        \end{column}
        \begin{column}{0.48\textwidth}
            \centering
            \textbf{\Large Aprendizaje}
            \begin{eqbox}
                \vspace*{-0.15in}
                \begin{equation*}
                    \Delta_L = \rho (\mathbf{rs}^T - \mathbf{sr}^T)
                \end{equation*}
            \end{eqbox}

            El término {\color{blue}antisimétrico} de \(W\) queda libre para {\color{blue}añadir patrones} robustos ante la degradación.
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{STDP}
    Método de aprendizaje basado en STDP (\emph{Spike-Timing-Dependent Plasticity}) en el que el {\color{VerdeEstimulo}estímulo} se añade a la actividad, \(\mathbf{x}\), no a las conexiones.
    
    \begin{eqbox}
        \begin{equation*}
            \Delta_L = \rho (\textbf{xy}^T - \textbf{yx}^T) \qquad \dot{\mathbf{y}} = \frac{1}{\tau}(\mathbf{x} - \mathbf{y}) \qquad \dot{\mathbf{x}} = W\mathbf{x} + {\color{VerdeEstimulo}\mathbf{h}(t)}
        \end{equation*}
    \end{eqbox}
\end{frame}

\begin{frame}{STDP}
    \textbf{Problema}: El estímulo empleado en \cite{susman2019} no es capaz de generar memorias de forma consistente.

    \textbf{Solución:} Para que el estímulo genere una perturbación antisimétrica significativa, este debe rotar en el plano del patrón con una frecuencia concreta.

    \begin{eqbox}
        \begin{equation*}
            \mathbf{h}(t) \simeq \sin(\omega t)\mathbf{r} + \cos(\omega t)\mathbf{s} \qquad \omega \tau \simeq 1
        \end{equation*}
    \end{eqbox}
\end{frame}
